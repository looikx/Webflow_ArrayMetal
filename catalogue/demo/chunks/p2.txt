function esc(s) {
return String(s ?? "")
.replace(/&/g, "&amp;").replace(/</g, "&lt;")

function parseHash() {
const h = (location.hash || "#/").replace(/^#\/?/, "");
const parts = h.split("/").filter(Boolean);
if (!parts.length) return { page: "home" };
if (parts[0] === "article" && parts[1]) return { page: "article", id: parts[1] };
if (parts[0] === "system" && parts[1]) return { page: "system", id: parts[1] };
if (groupById(parts[0])) return { page: "group", id: parts[0] };
return { page: "home" };
}

function filterArticles(list) {
const { systems, types, q } = state.filters;
return list.filter((a) => {
if (systems.size && !systems.has(a.sys)) return false;
if (types.size && !types.has(a.type)) return false;
if (q) {
const hay = (a.name + " " + a.code + " " + a.type + " " + a.sys).toLowerCase();
if (!hay.includes(q.toLowerCase())) return false;
}
return true;
});
}
function countBy(list, key) {
const m = {};
list.forEach((a) => { m[a[key]] = (m[a[key]] || 0) + 1; });
return m;
}

const app = document.getElementById("app");
function render() {
const route = parseHash();
if (route.page === "system" || route.page === "group") {
}
if (route.page === "home") {
state.filters = { systems: new Set(), types: new Set(), q: "" };
app.innerHTML = renderHome();
} else if (route.page === "group") {
state.filters = { systems: new Set(), types: new Set(), q: "" };
app.innerHTML = renderGroup(route.id);
} else if (route.page === "system") {
app.innerHTML = renderSystem(route.id);
bindSystemFilters(route.id);
} else if (route.page === "article") {
app.innerHTML = renderArticle(route.id);
} else {
app.innerHTML = renderHome();
}
bindNav();
window.scrollTo(0, 0);
}
function bindNav() {
app.querySelectorAll("[data-go]").forEach((el) => {
el.addEventListener("click", (e) => {
e.preventDefault();
const path = el.getAttribute("data-go");
if (path === "home") go("");
else go(path);
});
});
}

app.innerHTML = renderSystem(sysId);
bindSystemFilters(sysId);
bindNav();
});
});
