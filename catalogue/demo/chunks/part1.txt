function esc(s) {
return String(s ?? "")
.replace(/&/g, "&amp;").replace(/</g, "&lt;")
.replace(/>/g, "&gt;").replace(/"/g, "&quot;");
}
function groupById(id) { return GROUPS.find((g) => g.id === id); }
function articlesForSystem(sysId) { return ARTICLES.filter((a) => a.system === sysId); }
function articlesForGroup(groupId) {
const sysIds = groupById(groupId)?.systems || [];
return ARTICLES.filter((a) => sysIds.includes(a.system));
}
function parseHash() {
const h = (location.hash || "#/").replace(/^#\/?/, "");
const parts = h.split("/").filter(Boolean);
if (!parts.length) return { page: "home" };
if (parts[0] === "article" && parts[1]) return { page: "article", id: parts[1] };
if (parts[0] === "system" && parts[1]) return { page: "system", id: parts[1] };
if (groupById(parts[0])) return { page: "group", id: parts[0] };
return { page: "home" };
}
function go(path) {
location.hash = "#/" + path.replace(/^\//, "");
}

}
