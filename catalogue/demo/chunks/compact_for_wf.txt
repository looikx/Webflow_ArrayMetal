function bindFilters(sysId){app.querySelectorAll('input[data-filter="sys"]').forEach(function(input){input.addEventListener("change",function(){if(input.checked)state.filters.systems.add(input.value);else state.filters.systems.delete(input.value);app.innerHTML=system(sysId);bindFilters(sysId);bindNav();});});app.querySelectorAll('input[data-filter="type"]').forEach(function(input){input.addEventListener("change",function(){if(input.checked)state.filters.types.add(input.value);else state.filters.types.delete(input.value);app.innerHTML=system(sysId);bindFilters(sysId);bindNav();});});var q=document.getElementById("filter-q");if(q){q.addEventListener("input",function(){state.filters.q=q.value;var pos=q.selectionStart;app.innerHTML=system(sysId);bindFilters(sysId);bindNav();var nq=document.getElementById("filter-q");if(nq){nq.focus();nq.setSelectionRange(pos,pos);}});}var cl=document.getElementById("clear-filters");if(cl){cl.addEventListener("click",function(){state.filters={systems:new Set(),types:new Set(),q:""};app.innerHTML=system(sysId);bindFilters(sysId);bindNav();});}app.querySelectorAll("[data-remove-sys]").forEach(function(btn){btn.addEventListener("click",function(){state.filters.systems.delete(btn.getAttribute("data-remove-sys"));app.innerHTML=system(sysId);bindFilters(sysId);bindNav();});});app.querySelectorAll("[data-remove-type]").forEach(function(btn){btn.addEventListener("click",function(){state.filters.types.delete(btn.getAttribute("data-remove-type"));app.innerHTML=system(sysId);bindFilters(sysId);bindNav();});});app.querySelectorAll("[data-remove-q]").forEach(function(btn){btn.addEventListener("click",function(){state.filters.q="";app.innerHTML=system(sysId);bindFilters(sysId);bindNav();});});}
window.addEventListener("hashchange",render);
document.querySelectorAll("#am-cat .top-nav a, #am-cat .brand").forEach(function(a){a.addEventListener("click",function(e){var route=a.getAttribute("data-route");if(!route)return;e.preventDefault();if(route==="home")go("");else go(route);});});
render();
})();
</script>
