function esc(s) {
return String(s ?? "")
.replace(/&/g, "&amp;").replace(/</g, "&lt;")
.replace(/>/g, "&gt;").replace(/"/g, "&quot;");
}
function groupById(id) { return GROUPS.find((g) => g.id === id); }
function articlesForSystem(sysId) { return ARTICLES.filter((a) => a.system === sysId); }

const sysIds = groupById(groupId)?.systems || [];
return ARTICLES.filter((a) => sysIds.includes(a.system));
}
function parseHash() {
const h = (location.hash || "#/").replace(/^#\/?/, "");
const parts = h.split("/").filter(Boolean);
if (!parts.length) return { page: "home" };
if (parts[0] === "article" && parts[1]) return { page: "article", id: parts[1] };
if (parts[0] === "system" && parts[1]) return { page: "system", id: parts[1] };
if (groupById(parts[0])) return { page: "group", id: parts[0] };
return { page: "home" };
}
function go(path) {
location.hash = "#/" + path.replace(/^\//, "");
}

function filterArticles(list) {
const { systems, types, q } = state.filters;
return list.filter((a) => {
if (systems.size && !systems.has(a.sys)) return false;
if (types.size && !types.has(a.type)) return false;
if (q) {
const hay = (a.name + " " + a.code + " " + a.type + " " + a.sys).toLowerCase();
if (!hay.includes(q.toLowerCase())) return false;
}
return true;
});
}
function countBy(list, key) {
const m = {};
list.forEach((a) => { m[a[key]] = (m[a[key]] || 0) + 1; });
return m;
}

const app = document.getElementById("app");
function render() {
const route = parseHash();
if (route.page === "system" || route.page === "group") {
}
if (route.page === "home") {
state.filters = { systems: new Set(), types: new Set(), q: "" };
app.innerHTML = renderHome();
} else if (route.page === "group") {
state.filters = { systems: new Set(), types: new Set(), q: "" };
app.innerHTML = renderGroup(route.id);
} else if (route.page === "system") {
app.innerHTML = renderSystem(route.id);
bindSystemFilters(route.id);
} else if (route.page === "article") {
app.innerHTML = renderArticle(route.id);
} else {
app.innerHTML = renderHome();
}
bindNav();
window.scrollTo(0, 0);
}
function bindNav() {
app.querySelectorAll("[data-go]").forEach((el) => {
el.addEventListener("click", (e) => {
e.preventDefault();
const path = el.getAttribute("data-go");
if (path === "home") go("");
else go(path);
});
});
}
function bindSystemFilters(sysId) {
app.querySelectorAll('input[data-filter="sys"]').forEach((input) => {
input.addEventListener("change", () => {
if (input.checked) state.filters.systems.add(input.value);
else state.filters.systems.delete(input.value);
app.innerHTML = renderSystem(sysId);
bindSystemFilters(sysId);
bindNav();
});
});
app.querySelectorAll('input[data-filter="type"]').forEach((input) => {
input.addEventListener("change", () => {
if (input.checked) state.filters.types.add(input.value);
else state.filters.types.delete(input.value);
app.innerHTML = renderSystem(sysId);
bindSystemFilters(sysId);
bindNav();
});
});
const q = document.getElementById("filter-q");
if (q) {
q.addEventListener("input", () => {
state.filters.q = q.value;
const pos = q.selectionStart;
app.innerHTML = renderSystem(sysId);
bindSystemFilters(sysId);
bindNav();
const nq = document.getElementById("filter-q");
if (nq) { nq.focus(); nq.setSelectionRange(pos, pos); }
});
}
document.getElementById("clear-filters")?.addEventListener("click", () => {
state.filters = { systems: new Set(), types: new Set(), q: "" };
app.innerHTML = renderSystem(sysId);
bindSystemFilters(sysId);
bindNav();
});
app.querySelectorAll("[data-remove-sys]").forEach((btn) => {
btn.addEventListener("click", () => {
state.filters.systems.delete(btn.getAttribute("data-remove-sys"));
app.innerHTML = renderSystem(sysId);
bindSystemFilters(sysId);
bindNav();
});
});
app.querySelectorAll("[data-remove-type]").forEach((btn) => {
btn.addEventListener("click", () => {
state.filters.types.delete(btn.getAttribute("data-remove-type"));
app.innerHTML = renderSystem(sysId);
bindSystemFilters(sysId);
bindNav();
});
});
app.querySelectorAll("[data-remove-q]").forEach((btn) => {
btn.addEventListener("click", () => {
state.filters.q = "";
app.innerHTML = renderSystem(sysId);
bindSystemFilters(sysId);
bindNav();
});
});
}
window.addEventListener("hashchange", render);
document.querySelectorAll(".top-nav a, .brand").forEach((a) => {
a.addEventListener("click", (e) => {
const route = a.getAttribute("data-route");
if (!route) return;
e.preventDefault();
if (route === "home") go("");
else go(route);
});
});
render();
})();
